4 Trivy scan (fail on critical)

Dockerfile — Multi-Stage Build

# ---- Build stage ----
FROM python:3.12-slim AS builder
WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir --user -r requirements.txt

# ---- Final stage: no build tools, no root ----
FROM python:3.12-slim
WORKDIR /app

RUN useradd --create-home --uid 1000 appuser
COPY --from=builder /root/.local /home/appuser/.local
COPY --chown=appuser:appuser . .

ENV PATH=/home/appuser/.local/bin:$PATH \
    PYTHONUNBUFFERED=1

USER appuser
EXPOSE 8080
HEALTHCHECK --interval=30s --timeout=3s \
  CMD curl -f http://localhost:8080/health || exit 1

ENTRYPOINT ["python", "-m", "app"]

Why multi-stage: build tools (compilers, pip cache) never ship in the final image — smaller image, smaller attack surface. Non-root USER appuser + --chown on copy is the part interviewers listen for.
For a Go service the same principle goes further: FROM golang:1.22 AS builder → compile a static binary → COPY --from=builder /app/bin /bin/app into FROM scratch or gcr.io/distroless/static. No shell, no package manager, often <20MB final image.

CI/CD Pipeline — GitHub Actions + ArgoCD (GitOps)

git push main | GitHub Actions: test -> build -> Trivy scan -> push image (tag=SHA) | bump image tag in GitOps repo (git commit + push) | ArgoCD detects the diff (polls or webhook) | auto-sync: prune + selfHeal --> K8s cluster | canary rollout --> deploy annotation added to dashboard
.github/workflows/deploy.yml
name: build-and-deploy
on:
  push:
    branches: [main]

permissions:
  id-token: write   # OIDC, no static AWS keys
  contents: read

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Run tests
        run: pytest --maxfail=1

      - name: Configure AWS creds (OIDC)
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: arn:aws:iam::123456789012:role/gh-actions-deploy
          aws-region: us-east-1

      - name: Build, scan, push image
        run: |
          aws ecr get-login-password | \
            docker login --username AWS --password-stdin $ECR_REPO
          docker build -t $ECR_REPO:$GITHUB_SHA .
          trivy image --exit-code 1 \
            --severity CRITICAL,HIGH $ECR_REPO:$GITHUB_SHA
          docker push $ECR_REPO:$GITHUB_SHA

      - name: Bump tag in GitOps repo
        run: |
          git clone https://x-access-token:${{ secrets.GITOPS_TOKEN }}@github.com/fubo/gitops-repo.git
          cd gitops-repo
          yq -i '.image.tag = strenv(GITHUB_SHA)' apps/adserver/values.yaml
          git commit -am "deploy adserver@$GITHUB_SHA" && git push
argocd-app.yaml
apiVersion: argoproj.io/v1alpha1
kind: Application
metadata:
  name: adserver
  namespace: argocd
spec:
  project: default
  source:
    repoURL: https://github.com/fubo/gitops-repo.git
    targetRevision: main
    path: apps/adserver
    helm:
      valueFiles:
        - values.yaml
  destination:
    server: https://kubernetes.default.svc
    namespace: adtech
  syncPolicy:
    automated:
      prune: true       # remove what's no longer in git
      selfHeal: true    # revert manual kubectl drift
    syncOptions:
      - CreateNamespace=true
    retry:
      limit: 3
      backoff:
        duration: 10s
        maxDuration: 3m

CI (GitHub Actions) stops at "update the GitOps repo." CD (ArgoCD) takes over from there — that boundary is the whole point of GitOps: nothing ever pushes into the cluster directly.

Deployment.yaml — Production-Grade

apiVersion: apps/v1
kind: Deployment
metadata:
  name: adserver
  namespace: adtech
  labels: { app: adserver }
spec:
  replicas: 6
  strategy:
    type: RollingUpdate
    rollingUpdate:
      maxUnavailable: 0     # never drop below desired capacity
      maxSurge: 25%
  selector:
    matchLabels: { app: adserver }
  template:
    metadata:
      labels: { app: adserver }
    spec:
      serviceAccountName: adserver-sa       # IRSA-bound for AWS access
      topologySpreadConstraints:
        - maxSkew: 1
          topologyKey: topology.kubernetes.io/zone
          whenUnsatisfiable: DoNotSchedule
          labelSelector:
            matchLabels: { app: adserver }
      securityContext:
        runAsNonRoot: true
        runAsUser: 1000
        fsGroup: 1000
      containers:
        - name: adserver
          image: 123456789012.dkr.ecr.us-east-1.amazonaws.com/adserver:GITHUB_SHA
          ports:
            - containerPort: 8080
          resources:
            requests: { cpu: "500m", memory: "512Mi" }
            limits:   { cpu: "500m", memory: "512Mi" }   # requests=limits -> Guaranteed QoS
          readinessProbe:
            httpGet: { path: /health, port: 8080 }
            initialDelaySeconds: 5
            periodSeconds: 10
          livenessProbe:
            httpGet: { path: /health, port: 8080 }
            initialDelaySeconds: 15
            periodSeconds: 20
            failureThreshold: 3
          securityContext:
            allowPrivilegeEscalation: false
            readOnlyRootFilesystem: true
---
apiVersion: policy/v1
kind: PodDisruptionBudget
metadata:
  name: adserver-pdb
  namespace: adtech
spec:
  minAvailable: 80%
  selector:
    matchLabels: { app: adserver }

Five things an interviewer scans this for: maxUnavailable: 0 (zero-downtime), requests=limits (Guaranteed QoS), topologySpreadConstraints (real multi-AZ spread, not just replica count), non-root + readOnlyRootFilesystem (security posture), and the standalone PodDisruptionBudget — a Deployment alone doesn't protect you during a node drain, the PDB is what does.

SLI / SLO / Error Budget · Golden Signals

SLI = the measured indicator (e.g. success rate)
SLO = the target (e.g. 99.9%)
Error budget = 100% − SLO (0.1%) = allowed "badness"
Burn rate = speed of budget consumption vs sustainable pace

Golden signals: Latency (p95/p99) · Traffic · Errors · Saturation

HTTP & Exit Codes

502 upstream down/bad response
503 overloaded / health check failing
504 upstream too slow / timeout
429 rate limited

Exit codes: 0 ok · 1 general error · 126 not executable · 127 not found · 137 SIGKILL/OOM · 143 SIGTERM

One Terraform Module — VPC with Public/Private Subnets (HA NAT per AZ)

variables.tf
variable "name"     { type = string }
variable "vpc_cidr" { type = string
  default = "10.0.0.0/16" }
variable "azs" {
  type = list(string)          # ["us-east-1a","us-east-1b"]
}
variable "public_subnet_cidrs"  { type = list(string) }
variable "private_subnet_cidrs" { type = list(string) }
outputs.tf
output "vpc_id"             { value = aws_vpc.this.id }
output "public_subnet_ids"  {
  value = [for s in aws_subnet.public  : s.id] }
output "private_subnet_ids" {
  value = [for s in aws_subnet.private : s.id] }
usage
module "vpc" {
  source  = "./modules/vpc"
  name    = "fubo-adtech"
  azs     = ["us-east-1a", "us-east-1b"]
  public_subnet_cidrs  = ["10.0.1.0/24","10.0.2.0/24"]
  private_subnet_cidrs = ["10.0.11.0/24","10.0.12.0/24"]
}
main.tf
resource "aws_vpc" "this" {
  cidr_block = var.vpc_cidr
  enable_dns_support   = true
  enable_dns_hostnames = true
  tags = { Name = var.name }
}

resource "aws_internet_gateway" "this" {
  vpc_id = aws_vpc.this.id
}

resource "aws_subnet" "public" {
  for_each = { for i, az in var.azs :
    az => var.public_subnet_cidrs[i] }
  vpc_id = aws_vpc.this.id
  availability_zone       = each.key
  cidr_block               = each.value
  map_public_ip_on_launch = true
}

resource "aws_subnet" "private" {
  for_each = { for i, az in var.azs :
    az => var.private_subnet_cidrs[i] }
  vpc_id             = aws_vpc.this.id
  availability_zone  = each.key
  cidr_block         = each.value
}

# One EIP + NAT Gateway PER AZ = real HA
# (an AZ never depends on another AZ's NAT)
resource "aws_eip" "nat" {
  for_each = aws_subnet.public
  domain   = "vpc"
}

resource "aws_nat_gateway" "this" {
  for_each      = aws_subnet.public
  allocation_id = aws_eip.nat[each.key].id
  subnet_id     = each.value.id
}

resource "aws_route_table" "public" {
  vpc_id = aws_vpc.this.id
  route {
    cidr_block = "0.0.0.0/0"
    gateway_id = aws_internet_gateway.this.id
  }
}

resource "aws_route_table" "private" {
  for_each = aws_subnet.private
  vpc_id   = aws_vpc.this.id
  route {
    cidr_block     = "0.0.0.0/0"
    nat_gateway_id = aws_nat_gateway.this[each.key].id
  }
}

resource "aws_route_table_association" "public" {
  for_each       = aws_subnet.public
  subnet_id      = each.value.id
  route_table_id = aws_route_table.public.id
}

resource "aws_route_table_association" "private" {
  for_each       = aws_subnet.private
  subnet_id      = each.value.id
  route_table_id = aws_route_table.private[each.key].id
}

Say out loud: "each AZ's private subnet routes through its own AZ's NAT Gateway — one AZ losing its NAT never affects another AZ."
